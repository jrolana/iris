\chapter{Penetration Testing Script}
\label{app:penetration-testing-script}

This appendix presents the limited-scope penetration testing script used to evaluate selected security controls of the IRIS web application. The script was designed to test authentication, role-based access control, object-level authorization, Supabase RPC exposure, direct table access, file access, input validation, audit trail access, HTTP method handling, error handling, and optional rate-limit behavior. It automatically generates Markdown and CSV summaries, stores raw evidence files, and redacts configured tokens before the evidence files are saved.

The script is intended for controlled local or staging environments only. Before running the script, the required environment variables should be configured in \texttt{pentest.env}, \texttt{.env}, or \texttt{.env.local}.

\begin{lstlisting}[style=bashcode, caption={IRIS limited-scope penetration testing script}, label={lst:iris-pentest-script}]

#!/usr/bin/env bash

set -u

ENV_FILE="./pentest.env"
PROJECT_ENV_FILE="./.env"
PROJECT_ENV_LOCAL_FILE="./.env.local"
RESULTS_DIR="./pentest-results"
RAW_DIR="$RESULTS_DIR/raw"
TABLE_FILE="$RESULTS_DIR/pentest-summary.md"
CSV_FILE="$RESULTS_DIR/pentest-summary.csv"

mkdir -p "$RAW_DIR"

load_env_file() {
  local file="$1"

  if [ ! -f "$file" ]; then
    return
  fi

  set -a
  # shellcheck disable=SC1090
  source "$file"
  set +a
}

load_env_file "$PROJECT_ENV_FILE"
load_env_file "$PROJECT_ENV_LOCAL_FILE"

if [ -f "$ENV_FILE" ]; then
  load_env_file "$ENV_FILE"
fi

# =========================
# CONFIGURATION PLACEHOLDERS
# =========================

APP_URL="${APP_URL:-${NEXT_PUBLIC_SITE_URL:-http://localhost:3000}}"
SUPABASE_URL="${SUPABASE_URL:-${NEXT_PUBLIC_SUPABASE_URL:-https://YOUR_PROJECT_REF.supabase.co}}"
SUPABASE_ANON_KEY="${SUPABASE_ANON_KEY:-${NEXT_PUBLIC_SUPABASE_ANON_KEY:-YOUR_SUPABASE_ANON_KEY}}"

ADMIN_TOKEN="${ADMIN_TOKEN:-}"
TECHGEN_TOKEN="${TECHGEN_TOKEN:-}"
UPOFFICIAL_TOKEN="${UPOFFICIAL_TOKEN:-}"
INVALID_TOKEN="${INVALID_TOKEN:-invalid-token}"

OWN_APP_ID="${OWN_APP_ID:-}"
OTHER_APP_ID="${OTHER_APP_ID:-}"

OTHER_FILE_ID="${OTHER_FILE_ID:-}"

  echo
fi

# =========================
# HELPER FUNCTIONS
# =========================

timestamp() {
  date +"%Y-%m-%d %H:%M:%S"
}

safe_filename() {
  echo "$1" | tr ' /:?&=%' '________'
}

redact_file() {
  local file="$1"

  [ -n "${SUPABASE_ANON_KEY:-}" ] && sed -i.bak "s/${SUPABASE_ANON_KEY//\//\\/}/[REDACTED_SUPABASE_ANON_KEY]/g" "$file" 2>/dev/null || true
  [ -n "${ADMIN_TOKEN:-}" ] && sed -i.bak "s/${ADMIN_TOKEN//\//\\/}/[REDACTED_ADMIN_TOKEN]/g" "$file" 2>/dev/null || true
  [ -n "${TECHGEN_TOKEN:-}" ] && sed -i.bak "s/${TECHGEN_TOKEN//\//\\/}/[REDACTED_TECHGEN_TOKEN]/g" "$file" 2>/dev/null || true
  [ -n "${UPOFFICIAL_TOKEN:-}" ] && sed -i.bak "s/${UPOFFICIAL_TOKEN//\//\\/}/[REDACTED_UPOFFICIAL_TOKEN]/g" "$file" 2>/dev/null || true
  [ -n "${VALID_API_TOKEN:-}" ] && sed -i.bak "s/${VALID_API_TOKEN//\//\\/}/[REDACTED_API_TOKEN]/g" "$file" 2>/dev/null || true

  rm -f "$file.bak"
}

get_status_code() {
  local file="$1"
  grep -m 1 -E "^HTTP/[0-9.]+ [0-9]{3}" "$file" | awk '{print $2}'
}

contains_sensitive_user_fields() {
  local file="$1"
  grep -Eiq '"email"|"password"|"created_at"|"updated_at"|"google_id"|"avatar_url"|"last_sign_in"|"metadata"|"raw_user_meta_data"' "$file"
}

classify_denied_or_redirect() {
  local code="$1"

  case "$code" in
    301|302|303|307|308|401|403)
      echo "Passed"
      ;;
    200)
      echo "Needs Review"
      ;;
    "")
      echo "No Response"
      ;;
    *)
      echo "Needs Review"
      ;;
  esac
}

classify_allowed() {
  local code="$1"

  case "$code" in
    200)
      echo "Passed"
      ;;
    301|302|303|307|308|401|403)
      echo "Needs Review"
      ;;
    "")
      echo "No Response"
      ;;
    *)
      echo "Needs Review"
      ;;
  esac
}

classify_rejected() {
  local code="$1"

  case "$code" in
    400|401|403|404|405|415|422|429)
      echo "Passed"
      ;;
    200|201)
      echo "Needs Review"
      ;;
    "")
      echo "No Response"
      ;;
    *)
      echo "Needs Review"
      ;;
  esac
}

classify_method_rejected() {
  local code="$1"

  case "$code" in
    405|400|401|403|404)
      echo "Passed"
      ;;
    200|201|204)
      echo "Needs Review"
      ;;
    "")
      echo "No Response"
      ;;
    *)
      echo "Needs Review"
      ;;
  esac
}

classify_cors_restricted() {
  local code="$1"
  local file="$2"
  local forbidden_origin="$3"

  if [ -z "$code" ]; then
    echo "No Response"
  elif grep -Eiq "^access-control-allow-origin: \\*\\s*$" "$file"; then
    echo "Needs Review"
  elif grep -Eiq "^access-control-allow-origin: ${forbidden_origin}\\s*$" "$file"; then
    echo "Needs Review"
  else
    echo "Passed"
  fi
}

classify_rpc_safe() {
  local code="$1"
  local file="$2"

  if [ "$code" = "401" ] || [ "$code" = "403" ] || [ "$code" = "404" ]; then
    echo "Passed"
  elif [ "$code" = "200" ]; then
    if grep -Eq '\[\]|\{\}' "$file" && ! contains_sensitive_user_fields "$file"; then
      echo "Passed"
    else
      echo "Needs Review"
    fi
  elif [ -z "$code" ]; then
    echo "No Response"
  else
    echo "Needs Review"
  fi
}

add_result() {
  local area="$1"
  local test_name="$2"
  local expected="$3"
  local code="$4"
  local result="$5"
  local evidence="$6"
  local notes="$7"

  echo "| $area | $test_name | $expected | ${code:-N/A} | $result | $evidence | $notes |" >> "$TABLE_FILE"
  echo "\"$area\",\"$test_name\",\"$expected\",\"${code:-N/A}\",\"$result\",\"$evidence\",\"$notes\"" >> "$CSV_FILE"
}

run_curl_test() {
  local area="$1"
  local test_name="$2"
  local expected="$3"
  local classifier="$4"
  local evidence_name="$5"
  shift 5

  local evidence_file="$RAW_DIR/$(safe_filename "$evidence_name").txt"

  curl -k -i -sS "$@" > "$evidence_file" 2>&1
  redact_file "$evidence_file"

  local code
  code="$(get_status_code "$evidence_file")"

  local result
  case "$classifier" in
    denied)
      result="$(classify_denied_or_redirect "$code")"
      ;;
    allowed)
      result="$(classify_allowed "$code")"
      ;;
    rejected)
      result="$(classify_rejected "$code")"
      ;;
    method_rejected)
      result="$(classify_method_rejected "$code")"
      ;;
    cors_restricted)
      result="$(classify_cors_restricted "$code" "$evidence_file" "https://evil.example.com")"
      ;;
    rpc_safe)
      result="$(classify_rpc_safe "$code" "$evidence_file")"
      ;;
    *)
      result="Needs Review"
      ;;
  esac

  add_result "$area" "$test_name" "$expected" "$code" "$result" "$evidence_file" ""
}

require_value_warning() {
  local name="$1"
  local value="$2"

  if [ -z "$value" ] || echo "$value" | grep -q "YOUR_"; then
    echo "WARNING: $name is empty or still a placeholder. Some tests may be skipped or invalid."
  fi
}
